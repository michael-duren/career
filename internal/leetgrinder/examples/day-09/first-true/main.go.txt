package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func solve(n int64) int64 {
	lo, hi := int64(1), n
	format := func(x int64) string { return strconv.FormatInt(x, 10) }
	emit("start", "lo", format(lo), "hi", format(hi), "mid", "-", "blocks", "-", "enough", "-")
	for lo < hi {
		mid := lo + (hi-lo)/2
		blocks := mid * (mid + 1) / 2
		if blocks >= n {
			hi = mid
			emit("yes", "lo", format(lo), "hi", format(hi), "mid", format(mid), "blocks", format(blocks), "enough", "true")
		} else {
			lo = mid + 1
			emit("no", "lo", format(lo), "hi", format(hi), "mid", format(mid), "blocks", format(blocks), "enough", "false")
		}
	}

	emit("done", "lo", format(lo), "hi", format(hi), "mid", "-", "blocks", format(lo*(lo+1)/2), "enough", "true")
	return lo
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var n int64
	fmt.Fscan(in, &n)
	result(strconv.FormatInt(solve(n), 10))
}
