package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "strconv"
)

type variable struct { Name string `json:"name"`; Value string `json:"value"` }
type eventRecord struct { Event string `json:"event"`; Variables []variable `json:"variables"` }
type resultRecord struct { Result string `json:"result"` }
func emit(event string, i int, current, state, answer string) {
    record := eventRecord{event, []variable{{"i",strconv.Itoa(i)},{"current",current},{"state",state},{"answer",answer}}}
    data, _ := json.Marshal(record); fmt.Println(string(data))
}
func result(answer string) { data,_:=json.Marshal(resultRecord{answer});fmt.Println(string(data)) }
func main() {
    in:=bufio.NewReader(os.Stdin)
    board:=make([]string,9);for i:=range board {fmt.Fscan(in,&board[i])}
    var rows,cols,boxes [9]map[byte]bool
    for i:=0;i<9;i++ {rows[i]=make(map[byte]bool);cols[i]=make(map[byte]bool);boxes[i]=make(map[byte]bool)}
    step:=0;answer:="true"
    emit("start",-1,"-","checked=0","pending")
    for r:=0;r<9 && answer=="true";r++ {for c:=0;c<9;c++ {
        digit:=board[r][c];if digit=='.' {continue}
        box:=(r/3)*3+c/3;current:=string(digit)+"@"+strconv.Itoa(r)+","+strconv.Itoa(c)
        if rows[r][digit] || cols[c][digit] || boxes[box][digit] {
            answer="false";emit("conflict",step,current,"checked="+strconv.Itoa(step),answer);break
        }
        rows[r][digit]=true;cols[c][digit]=true;boxes[box][digit]=true;step++
        emit("check",step-1,current,"checked="+strconv.Itoa(step),"pending")
    }}
    emit("done",step,"-","checked="+strconv.Itoa(step),answer)
    result(answer)
}
