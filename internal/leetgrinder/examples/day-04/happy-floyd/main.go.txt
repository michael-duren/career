package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "strconv"
)

type variable struct { Name string `json:"name"`; Value string `json:"value"` }
type eventRecord struct { Event string `json:"event"`; Variables []variable `json:"variables"` }
type resultRecord struct { Result string `json:"result"` }
func emit(event string, i int, current, state, answer string) {
    record := eventRecord{event, []variable{{"i",strconv.Itoa(i)},{"current",current},{"state",state},{"answer",answer}}}
    data, _ := json.Marshal(record); fmt.Println(string(data))
}
func result(answer string) { data,_:=json.Marshal(resultRecord{answer});fmt.Println(string(data)) }
func digitSquare(value int) int {total:=0;for value>0 {d:=value%10;total+=d*d;value/=10};return total}
func main() {
    in:=bufio.NewReader(os.Stdin)
    var value int;fmt.Fscan(in,&value);slow,fast,step:=value,value,0
    state:=func()string{return "slow="+strconv.Itoa(slow)+";fast="+strconv.Itoa(fast)}
    emit("start",-1,strconv.Itoa(value),state(),"pending")
    for {
        slow=digitSquare(slow);fast=digitSquare(digitSquare(fast))
        emit("step",step,strconv.Itoa(slow),state(),"pending")
        step++
        if slow==1 || fast==1 || slow==fast {break}
    }
    answer:="false";if slow==1 || fast==1 {answer="true"}
    emit("done",step,strconv.Itoa(slow),state(),answer)
    result(answer)
}
