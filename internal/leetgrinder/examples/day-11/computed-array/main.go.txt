package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func solve(n, target int) int {
	lo, hi := 0, n
	emit("start", "lo", itoa(lo), "hi", itoa(hi), "mid", "-", "value", "-")
	for lo <= hi {
		mid := lo + (hi-lo)/2
		value := mid * mid
		if value == target {
			emit("found", "lo", itoa(lo), "hi", itoa(hi), "mid", itoa(mid), "value", itoa(value))
			return mid
		}
		if value < target {
			lo = mid + 1
			emit("go-right", "lo", itoa(lo), "hi", itoa(hi), "mid", itoa(mid), "value", itoa(value))
		} else {
			hi = mid - 1
			emit("go-left", "lo", itoa(lo), "hi", itoa(hi), "mid", itoa(mid), "value", itoa(value))
		}
	}

	emit("absent", "lo", itoa(lo), "hi", itoa(hi), "mid", "-", "value", "-")
	return -1
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var n, target int
	fmt.Fscan(in, &n, &target)
	result(itoa(solve(n, target)))
}
