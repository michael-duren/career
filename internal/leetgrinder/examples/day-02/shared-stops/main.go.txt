package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func solve(first, second []int) []int {
	stops := map[int]bool{}
	position := 0
	for _, length := range first[:len(first)-1] {
		position += length
		stops[position] = true
		emit("stop", "route", "A", "length", itoa(length), "position", itoa(position), "shared", "[]")
	}
	shared := []int{}
	position = 0
	for _, length := range second[:len(second)-1] {
		position += length
		if stops[position] {
			shared = append(shared, position)
		}
		emit("check", "route", "B", "length", itoa(length), "position", itoa(position), "shared", show(shared))
	}

	emit("done", "route", "-", "length", "-", "position", "-", "shared", show(shared))
	return shared
}

func parse(line string) []int {
	values := []int{}
	for _, field := range strings.Fields(line) {
		value, _ := strconv.Atoi(field)
		values = append(values, value)
	}
	return values
}

func main() {
	scanner := bufio.NewScanner(os.Stdin)
	scanner.Scan()
	first := parse(scanner.Text())
	scanner.Scan()
	second := parse(scanner.Text())
	result(show(solve(first, second)))
}
