package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, fields ...string) {
	variables := make([]variable, 0, len(fields)/2)
	for i := 0; i < len(fields); i += 2 {
		variables = append(variables, variable{fields[i], fields[i+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}
func result(value string) {
	data, _ := json.Marshal(map[string]string{"result": value})
	fmt.Println(string(data))
}
func s(n int) string { return strconv.Itoa(n) }

func solve(values []int) {
	answer := "none"
	candidate, balance := 0, 0
	emit("start", "i", "-1", "value", "-", "candidate", "-", "balance", "0", "seen", "0")
	for i, value := range values {
		if balance == 0 {
			candidate = value
			balance = 1
		} else if candidate == value {
			balance++
		} else {
			balance--
		}
		emit("vote", "i", s(i), "value", s(value), "candidate", s(candidate), "balance", s(balance), "seen", "0")
	}
	seen := 0
	for i, value := range values {
		if value == candidate {
			seen++
		}
		emit("verify", "i", s(i), "value", s(value), "candidate", s(candidate), "balance", s(balance), "seen", s(seen))
	}
	if seen > len(values)/2 {
		answer = s(candidate)
	}
	emit("done", "i", s(len(values)), "value", "-", "candidate", s(candidate), "balance", s(balance), "seen", s(seen))
	result(answer)
}
func main() {
	in := bufio.NewReader(os.Stdin)
	var n int
	fmt.Fscan(in, &n)
	values := make([]int, n)
	for i := range values {
		fmt.Fscan(in, &values[i])
	}
	solve(values)
}
