package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, fields ...string) {
	variables := make([]variable, 0, len(fields)/2)
	for i := 0; i < len(fields); i += 2 {
		variables = append(variables, variable{fields[i], fields[i+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}
func result(value string) {
	data, _ := json.Marshal(map[string]string{"result": value})
	fmt.Println(string(data))
}
func s(n int) string { return strconv.Itoa(n) }

func countState(counts []int) string {
	parts := []string{}
	for i, n := range counts {
		if n != 0 {
			parts = append(parts, string(rune('a'+i))+":"+s(n))
		}
	}
	return "{" + strings.Join(parts, ",") + "}"
}
func solve(note, magazine string) {
	ok := true
	counts := make([]int, 26)
	emit("start", "i", "-1", "letter", "-", "position", "-1", "state", "{}", "ok", "true")
	for j := 0; j < len(magazine); j++ {
		letter := magazine[j]
		counts[letter-'a']++
		emit("supply", "i", s(j), "letter", string(letter), "position", s(j), "state", countState(counts), "ok", "true")
	}
	for i := 0; i < len(note); i++ {
		letter := note[i]
		counts[letter-'a']--
		ok = counts[letter-'a'] >= 0
		emit("request", "i", s(i), "letter", string(letter), "position", "-1", "state", countState(counts), "ok", strconv.FormatBool(ok))
		if !ok {
			break
		}
	}
	emit("done", "i", s(len(note)), "letter", "-", "position", "-1", "state", countState(counts), "ok", strconv.FormatBool(ok))
	result(strconv.FormatBool(ok))
}
func main() {
	in := bufio.NewReader(os.Stdin)
	var note, magazine string
	fmt.Fscan(in, &note, &magazine)
	solve(note, magazine)
}
