package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, fields ...string) {
	variables := make([]variable, 0, len(fields)/2)
	for i := 0; i < len(fields); i += 2 {
		variables = append(variables, variable{fields[i], fields[i+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}
func result(value string) {
	data, _ := json.Marshal(map[string]string{"result": value})
	fmt.Println(string(data))
}
func s(n int) string { return strconv.Itoa(n) }

func solve(values []int) {
	answer := "none"
	lastSeen := 0
	lastCandidate := 0
	emit("start", "i", "-1", "candidate", "-", "tally", "0", "answer", "none")
	for i, candidate := range values {
		seen := 0
		for _, value := range values {
			if value == candidate {
				seen++
			}
		}
		lastSeen = seen
		lastCandidate = candidate
		emit("test", "i", s(i), "candidate", s(candidate), "tally", s(seen), "answer", "none")
		if seen > len(values)/2 {
			answer = s(candidate)
			break
		}
	}
	emit("done", "i", s(len(values)), "candidate", s(lastCandidate), "tally", s(lastSeen), "answer", answer)
	result(answer)
}
func main() {
	in := bufio.NewReader(os.Stdin)
	var n int
	fmt.Fscan(in, &n)
	values := make([]int, n)
	for i := range values {
		fmt.Fscan(in, &values[i])
	}
	solve(values)
}
