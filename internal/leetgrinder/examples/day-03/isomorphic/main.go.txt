package main
import (
    "fmt"
    "os"
)
func emit(event string, i int, left, right string, pairs int, valid bool) {
    fmt.Printf("{\"event\":\"%s\",\"variables\":[{\"name\":\"i\",\"value\":\"%d\"},{\"name\":\"left\",\"value\":\"%s\"},{\"name\":\"right\",\"value\":\"%s\"},{\"name\":\"pairs\",\"value\":\"%d\"},{\"name\":\"valid\",\"value\":\"%t\"}]}\n", event, i, left, right, pairs, valid)
}
func result(value bool) { fmt.Printf("{\"result\":\"%t\"}\n", value) }

func main() {
    var left, right string; fmt.Fscan(os.Stdin, &left, &right)
    a := []string{}; b := []string{}; for _, c := range left { a = append(a, string(c)) }; for _, c := range right { b = append(b, string(c)) }
    forward, reverse := map[string]string{}, map[string]string{}
    emit("start", -1, "-", "-", 0, true)
    if len(a) != len(b) { emit("done", len(a), "-", "-", 0, false); result(false); return }
    for i, x := range a {
        y := b[i]
        oldY, hasX := forward[x]
        oldX, hasY := reverse[y]
        if (hasX && oldY != y) || (hasY && oldX != x) {
            emit("reject", i, x, y, len(forward), false)
            emit("done", len(a), "-", "-", len(forward), false)
            result(false); return
        }
        forward[x], reverse[y] = y, x
        emit("pair", i, x, y, len(forward), true)
    }
    emit("done", len(a), "-", "-", len(forward), true)
    result(true)
}
