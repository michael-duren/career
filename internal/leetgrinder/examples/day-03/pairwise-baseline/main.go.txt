package main
import (
    "bufio"
    "fmt"
    "os"
    "strings"
)
func emit(event string, i int, left, right string, pairs int, valid bool) {
    fmt.Printf("{\"event\":\"%s\",\"variables\":[{\"name\":\"i\",\"value\":\"%d\"},{\"name\":\"left\",\"value\":\"%s\"},{\"name\":\"right\",\"value\":\"%s\"},{\"name\":\"pairs\",\"value\":\"%d\"},{\"name\":\"valid\",\"value\":\"%t\"}]}\n", event, i, left, right, pairs, valid)
}
func result(value bool) { fmt.Printf("{\"result\":\"%t\"}\n", value) }

func main() {
    scanner:=bufio.NewScanner(os.Stdin); scanner.Scan(); left:=scanner.Text(); scanner.Scan(); right:=scanner.Text()
    a:=strings.Split(left,""); b:=strings.Fields(right)
    checks:=0
    emit("start",-1,"-","-",checks,true)
    if len(a)!=len(b) { emit("done",len(a),"-","-",checks,false); result(false); return }
    for i:=range a {
        for j:=0;j<i;j++ {
            checks++
            if (a[i]==a[j]) != (b[i]==b[j]) {
                emit("reject",i,fmt.Sprintf("i=%d,j=%d %s/%s",i,j,a[i],a[j]),fmt.Sprintf("i=%d,j=%d %s/%s",i,j,b[i],b[j]),checks,false)
                emit("done",len(a),"-","-",checks,false)
                result(false); return
            }
            emit("compare",i,fmt.Sprintf("i=%d,j=%d %s/%s",i,j,a[i],a[j]),fmt.Sprintf("i=%d,j=%d %s/%s",i,j,b[i],b[j]),checks,true)
        }
    }
    emit("done",len(a),"-","-",checks,true)
    result(true)
}
