package main

import (
    "encoding/json"
    "fmt"
    "strings"
)

type variable struct { Name string `json:"name"`; Value string `json:"value"` }
type eventRecord struct { Event string `json:"event"`; Variables []variable `json:"variables"` }
type resultRecord struct { Result string `json:"result"` }

func field(name, value string) variable { return variable{Name: name, Value: value} }
func number(value int) string { return fmt.Sprint(value) }

func emit(event string, fields ...variable) {
    data, _ := json.Marshal(eventRecord{Event: event, Variables: fields})
    fmt.Println(string(data))
}

func shown(counts [60]int) string {
    parts := []string{}
    for r, count := range counts {
        if count != 0 { parts = append(parts, number(r)+":"+number(count)) }
    }
    return "{" + strings.Join(parts, ",") + "}"
}

func listText(values []int) string {
    parts := make([]string, len(values))
    for i, value := range values { parts[i] = number(value) }
    return "[" + strings.Join(parts, ",") + "]"
}

func main() {
    var size int
    if _, err := fmt.Scan(&size); err != nil { panic(err) }
    songs := make([]int, size)
    for i := range songs { fmt.Scan(&songs[i]) }
    var counts [60]int
    pairs := 0
    emit("start", field("songs", listText(songs)), field("counts", shown(counts)), field("pairs", number(pairs)))
    for i, duration := range songs {
        remainder := duration % 60
        need := (60 - remainder) % 60
        matches := counts[need]
        emit("lookup", field("i", number(i)), field("duration", number(duration)), field("remainder", number(remainder)), field("need", number(need)), field("matches", number(matches)), field("pairs", number(pairs)), field("counts", shown(counts)))
        pairs += matches
        emit("count", field("i", number(i)), field("need", number(need)), field("matches", number(matches)), field("pairs", number(pairs)), field("counts", shown(counts)))
        counts[remainder]++
        emit("store", field("i", number(i)), field("remainder", number(remainder)), field("pairs", number(pairs)), field("counts", shown(counts)))
    }
    emit("done", field("pairs", number(pairs)), field("counts", shown(counts)))
    data, _ := json.Marshal(resultRecord{Result: number(pairs)})
    fmt.Println(string(data))
}
