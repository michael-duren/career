package main

import (
    "bufio"
    "fmt"
    "os"
    "strings"
)

type field struct { name, value string }

func shown(counts [26]int) string {
    parts := []string{}
    for i, count := range counts {
        if count != 0 { parts = append(parts, fmt.Sprintf("%c:%d", 'a' + i, count)) }
    }
    return "{" + strings.Join(parts, ",") + "}"
}

func emit(event string, fields ...field) {
    fmt.Printf("{\"event\":\"%s\",\"variables\":[", event)
    for i, f := range fields {
        if i > 0 { fmt.Print(",") }
        fmt.Printf("{\"name\":\"%s\",\"value\":\"%s\"}", f.name, f.value)
    }
    fmt.Println("]}")
}

func main() {
    input := bufio.NewScanner(os.Stdin)
    input.Scan()
    first := input.Text()
    input.Scan()
    second := input.Text()
    var counts [26]int
    emit("start", field{"first", first}, field{"second", second}, field{"counts", shown(counts)})
    for i, letter := range first {
        counts[letter-'a']++
        emit("add", field{"i", fmt.Sprint(i)}, field{"letter", string(letter)}, field{"counts", shown(counts)})
    }
    answer := len(first) == len(second)
    if answer {
        for i, letter := range second {
            counts[letter-'a']--
            emit("remove", field{"i", fmt.Sprint(i)}, field{"letter", string(letter)}, field{"counts", shown(counts)})
            if counts[letter-'a'] < 0 {
                answer = false
                break
            }
        }
    }
    emit("done", field{"counts", shown(counts)}, field{"anagram", fmt.Sprint(answer)})
    fmt.Printf("{\"result\":\"%t\"}\n", answer)
}
