package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func showCounts(counts map[int]int) string {
	text := strings.TrimSuffix(strings.TrimPrefix(fmt.Sprint(counts), "map["), "]")
	return "{" + strings.ReplaceAll(text, " ", ",") + "}"
}

func solve(nums []int, target int) int {
	counts := map[int]int{}
	pairs := 0
	emit("start", "i", "-", "value", "-", "need", "-", "matches", "-", "pairs", itoa(pairs), "counts", showCounts(counts))
	for i, value := range nums {
		need := target - value
		matches := counts[need]
		pairs += matches
		emit("count", "i", itoa(i), "value", itoa(value), "need", itoa(need), "matches", itoa(matches), "pairs", itoa(pairs), "counts", showCounts(counts))
		counts[value]++
		emit("store", "i", itoa(i), "value", itoa(value), "need", itoa(need), "matches", itoa(matches), "pairs", itoa(pairs), "counts", showCounts(counts))
	}

	emit("done", "i", "-", "value", "-", "need", "-", "matches", "-", "pairs", itoa(pairs), "counts", showCounts(counts))
	return pairs
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var n, target int
	fmt.Fscan(in, &n, &target)
	nums := make([]int, n)
	for k := range nums {
		fmt.Fscan(in, &nums[k])
	}
	result(itoa(solve(nums, target)))
}
