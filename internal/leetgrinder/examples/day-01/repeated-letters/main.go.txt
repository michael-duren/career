package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func solve(word string) string {
	var counts [26]int
	emit("start", "i", "-", "letter", "-", "count", "-", "repeated", "")
	for i := 0; i < len(word); i++ {
		letter := word[i]
		counts[letter-'a']++
		emit("count", "i", itoa(i), "letter", string(letter), "count", itoa(counts[letter-'a']), "repeated", "")
	}
	repeated := ""
	for k := 0; k < 26; k++ {
		if counts[k] > 1 {
			repeated += string(rune('a' + k))
			emit("repeat", "i", "-", "letter", string(rune('a'+k)), "count", itoa(counts[k]), "repeated", repeated)
		}
	}

	emit("done", "i", "-", "letter", "-", "count", "-", "repeated", repeated)
	if repeated == "" {
		return "none"
	}
	return repeated
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var word string
	fmt.Fscan(in, &word)
	result(solve(word))
}
