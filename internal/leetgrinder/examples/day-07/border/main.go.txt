package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func showGrid(grid [][]int) string {
	parts := make([]string, len(grid))
	for r, row := range grid {
		parts[r] = show(row)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func solve(grid [][]int) []int {
	rows, cols := len(grid), len(grid[0])
	order := []int{}
	emit("start", "side", "-", "order", show(order))
	for c := 0; c < cols; c++ {
		order = append(order, grid[0][c])
	}
	emit("top", "side", "top", "order", show(order))
	for r := 1; r < rows; r++ {
		order = append(order, grid[r][cols-1])
	}
	emit("right", "side", "right", "order", show(order))
	if rows > 1 {
		for c := cols - 2; c >= 0; c-- {
			order = append(order, grid[rows-1][c])
		}
		emit("bottom", "side", "bottom", "order", show(order))
	}
	if cols > 1 {
		for r := rows - 2; r > 0; r-- {
			order = append(order, grid[r][0])
		}
		emit("left", "side", "left", "order", show(order))
	}

	emit("done", "side", "-", "order", show(order))
	return order
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var rows, cols int
	fmt.Fscan(in, &rows, &cols)
	grid := make([][]int, rows)
	for r := range grid {
		grid[r] = make([]int, cols)
		for c := range grid[r] {
			fmt.Fscan(in, &grid[r][c])
		}
	}
	result(show(solve(grid)))
}
