package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func solve(a, b []int) []int {
	merged := []int{}
	i, j := 0, 0
	emit("start", "i", itoa(i), "j", itoa(j), "merged", show(merged))
	for i < len(a) || j < len(b) {
		if j == len(b) || (i < len(a) && a[i] <= b[j]) {
			merged = append(merged, a[i])
			i++
			emit("take-a", "i", itoa(i), "j", itoa(j), "merged", show(merged))
		} else {
			merged = append(merged, b[j])
			j++
			emit("take-b", "i", itoa(i), "j", itoa(j), "merged", show(merged))
		}
	}

	emit("done", "i", itoa(i), "j", itoa(j), "merged", show(merged))
	return merged
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var n, m int
	fmt.Fscan(in, &n)
	a := make([]int, n)
	for k := range a {
		fmt.Fscan(in, &a[k])
	}
	fmt.Fscan(in, &m)
	b := make([]int, m)
	for k := range b {
		fmt.Fscan(in, &b[k])
	}
	result(show(solve(a, b)))
}
