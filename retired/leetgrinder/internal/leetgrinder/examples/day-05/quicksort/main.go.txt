package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func quicksort(values []int, lo, hi int) {
	if lo >= hi {
		return
	}
	pivot := values[hi]
	store := lo
	emit("pivot", "lo", itoa(lo), "hi", itoa(hi), "pivot", itoa(pivot), "store", itoa(store), "scan", "-", "values", show(values))
	for scan := lo; scan < hi; scan++ {
		if values[scan] < pivot {
			values[store], values[scan] = values[scan], values[store]
			store++
			emit("smaller", "lo", itoa(lo), "hi", itoa(hi), "pivot", itoa(pivot), "store", itoa(store), "scan", itoa(scan), "values", show(values))
		} else {
			emit("not-smaller", "lo", itoa(lo), "hi", itoa(hi), "pivot", itoa(pivot), "store", itoa(store), "scan", itoa(scan), "values", show(values))
		}
	}
	values[store], values[hi] = values[hi], values[store]
	emit("place", "lo", itoa(lo), "hi", itoa(hi), "pivot", itoa(pivot), "store", itoa(store), "scan", "-", "values", show(values))
	quicksort(values, lo, store-1)
	quicksort(values, store+1, hi)
}

func solve(values []int) []int {
	last := len(values) - 1
	emit("start", "lo", "0", "hi", itoa(last), "pivot", "-", "store", "-", "scan", "-", "values", show(values))
	quicksort(values, 0, last)

	emit("done", "lo", "0", "hi", itoa(last), "pivot", "-", "store", "-", "scan", "-", "values", show(values))
	return values
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var n int
	fmt.Fscan(in, &n)
	values := make([]int, n)
	for k := range values {
		fmt.Fscan(in, &values[k])
	}
	result(show(solve(values)))
}
