package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func solve(values []int) []int {
	emit("start", "i", "-", "key", "-", "j", "-", "values", show(values))
	for i := 1; i < len(values); i++ {
		key := values[i]
		j := i - 1
		emit("take", "i", itoa(i), "key", itoa(key), "j", itoa(j), "values", show(values))
		for j >= 0 && values[j] > key {
			values[j+1] = values[j]
			emit("shift", "i", itoa(i), "key", itoa(key), "j", itoa(j), "values", show(values))
			j--
		}
		values[j+1] = key
		emit("place", "i", itoa(i), "key", itoa(key), "j", itoa(j), "values", show(values))
	}

	emit("done", "i", itoa(len(values)), "key", "-", "j", "-", "values", show(values))
	return values
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var n int
	fmt.Fscan(in, &n)
	values := make([]int, n)
	for k := range values {
		fmt.Fscan(in, &values[k])
	}
	result(show(solve(values)))
}
