package main
import("bufio";"fmt";"os";"strconv";"strings")
func flat(a []int)string{
    parts:=make([]string,len(a));for i,x:=range a{parts[i]=strconv.Itoa(x)}
    return "["+strings.Join(parts,",")+"]"
}
func nested(a [][]int)string{
    parts:=make([]string,len(a));for i,row:=range a{parts[i]=flat(row)}
    return "["+strings.Join(parts,",")+"]"
}
func emit(event string,depth,visits int,level []int,result [][]int){
    fmt.Printf("{\"event\":\"%s\",\"variables\":[{\"name\":\"depth\",\"value\":\"%d\"},{\"name\":\"visits\",\"value\":\"%d\"},{\"name\":\"level\",\"value\":\"%s\"},{\"name\":\"result\",\"value\":\"%s\"}]}\n",event,depth,visits,flat(level),nested(result))
}
func height(node int,left,right []int)int{
    if node==-1{return 0}
    a:=height(left[node],left,right);b:=height(right[node],left,right)
    if a>b{return a+1};return b+1
}
func collect(node,remaining int,value,left,right []int,level *[]int)int{
    if node==-1{return 0}
    if remaining==1{*level=append(*level,value[node]);return 1}
    return 1+collect(left[node],remaining-1,value,left,right,level)+collect(right[node],remaining-1,value,left,right,level)
}
func main(){
    in:=bufio.NewReader(os.Stdin);var n int;fmt.Fscan(in,&n)
    value:=make([]int,n);left:=make([]int,n);right:=make([]int,n)
    for i:=0;i<n;i++{fmt.Fscan(in,&value[i],&left[i],&right[i])}
    result:=[][]int{};visits:=0 // MODE: rescan
    emit("init",0,visits,[]int{},result)
    maximum:=0;if n>0{maximum=height(0,left,right)}
    emit("height",maximum,visits,[]int{},result)
    for depth:=1;depth<=maximum;depth++{
        level:=[]int{}
        emit("pass_start",depth,visits,level,result)
        visits+=collect(0,depth,value,left,right,&level)
        result=append(result,level)
        emit("pass_end",depth,visits,level,result)
    }
    emit("done",maximum,visits,[]int{},result)
    fmt.Printf("{\"result\":\"%s\"}\n",nested(result))
}
