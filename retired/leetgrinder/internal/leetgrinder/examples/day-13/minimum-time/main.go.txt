package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type variable struct {
	Name  string `json:"name"`
	Value string `json:"value"`
}

func emit(event string, pairs ...string) {
	variables := []variable{}
	for k := 0; k+1 < len(pairs); k += 2 {
		variables = append(variables, variable{pairs[k], pairs[k+1]})
	}
	data, _ := json.Marshal(struct {
		Event     string     `json:"event"`
		Variables []variable `json:"variables"`
	}{event, variables})
	fmt.Println(string(data))
}

func result(answer string) {
	data, _ := json.Marshal(struct {
		Result string `json:"result"`
	}{answer})
	fmt.Println(string(data))
}

func itoa(value int) string { return strconv.Itoa(value) }

func show(values []int) string {
	parts := make([]string, len(values))
	for k, value := range values {
		parts[k] = strconv.Itoa(value)
	}
	return "[" + strings.Join(parts, ",") + "]"
}

func tripsBy(time int, rates []int) int {
	trips := 0
	for _, rate := range rates {
		trips += time / rate
	}
	return trips
}

func solve(rates []int, needed int) int {
	fastest := rates[0]
	for _, rate := range rates {
		fastest = min(fastest, rate)
	}
	lo, hi := 1, fastest*needed
	emit("start", "lo", itoa(lo), "hi", itoa(hi), "mid", "-", "trips", "-")
	for lo < hi {
		mid := lo + (hi-lo)/2
		trips := tripsBy(mid, rates)
		if trips >= needed {
			hi = mid
			emit("enough", "lo", itoa(lo), "hi", itoa(hi), "mid", itoa(mid), "trips", itoa(trips))
		} else {
			lo = mid + 1
			emit("short", "lo", itoa(lo), "hi", itoa(hi), "mid", itoa(mid), "trips", itoa(trips))
		}
	}

	emit("done", "lo", itoa(lo), "hi", itoa(hi), "mid", "-", "trips", itoa(tripsBy(lo, rates)))
	return lo
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var n, needed int
	fmt.Fscan(in, &n, &needed)
	rates := make([]int, n)
	for k := range rates {
		fmt.Fscan(in, &rates[k])
	}
	result(itoa(solve(rates, needed)))
}
